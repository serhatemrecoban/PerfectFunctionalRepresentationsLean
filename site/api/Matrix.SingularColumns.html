Equal values and patterns on the same active output labels determine a channel.
∀ {α : Type u} {β : Type v} [inst : Fintype.{u} α] (κ η : α → PMF.{v} β)
  (hκ : PerfectFunctionalRepresentations.IsSingularChannel.{u, v} κ)
  (hη : PerfectFunctionalRepresentations.IsSingularChannel.{u, v} η)
  (ha : ∀ (y : β), Membership.mem.{v, v} (PerfectFunctionalRepresentations.activeOutputs.{u, v} κ) y)
  (hb : ∀ (y : β), Membership.mem.{v, v} (PerfectFunctionalRepresentations.activeOutputs.{u, v} η) y),
  (∀ (y : β),
      Eq.{u + 1} (PerfectFunctionalRepresentations.columnPattern.{u, v} κ (Subtype.mk.{v + 1} y (ha y)))
        (PerfectFunctionalRepresentations.columnPattern.{u, v} η (Subtype.mk.{v + 1} y (hb y)))) →
    (∀ (y : β),
        Eq.{1} (PerfectFunctionalRepresentations.columnValue.{u, v} κ hκ (Subtype.mk.{v + 1} y (ha y)))
          (PerfectFunctionalRepresentations.columnValue.{u, v} η hη (Subtype.mk.{v + 1} y (hb y)))) →
      Eq.{max (u + 1) (v + 1)} κ η
